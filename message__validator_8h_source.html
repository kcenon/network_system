152 [[nodiscard]] static size_t safe_copy(
153 void* dest,
154 size_t dest_size,
155 const void* src,
156 size_t src_size) noexcept {
157 if (!dest || !src || dest_size == 0 || src_size == 0) {
158 return 0;
159 }
160
161 size_t copy_size = std::min(dest_size, src_size);
162 std::memcpy(dest, src, copy_size);
163 return copy_size;
164 }
165
174 [[nodiscard]] static size_t safe_strcpy(
175 char* dest,
176 size_t dest_size,
177 const char* src) noexcept {
178 if (!dest || dest_size == 0 || !src) {
179 return 0;
180 }
181
182 size_t src_len = std::strlen(src);
183 size_t copy_len = std::min(dest_size - 1, src_len);
184
185 std::memcpy(dest, src, copy_len);
186 dest[copy_len] = '\0';
187
188 return copy_len;
189 }

202 [[nodiscard]] static validation_result validate_http_header(
203 std::string_view header) noexcept {
204 // Check size
205 if (header.size() > message_limits::MAX_HEADER_SIZE) {
206 return validation_result::size_exceeded;
207 }
208
209 // Check for NULL bytes (NULL byte injection attack)
210 if (header.find('\0') != std::string_view::npos) {
211 return validation_result::null_byte_detected;
212 }
213
214 // Check for invalid control characters (except \r\n\t)
215 for (char c : header) {
216 if (c < 0x20 && c != '\r' && c != '\n' && c != '\t') {
217 return validation_result::invalid_character;
218 }
219 }
220
221 return validation_result::ok;
222 }
223
230 [[nodiscard]] static bool validate_header_count(size_t count) noexcept {
231 return count <= message_limits::MAX_HEADER_COUNT;
232 }
233
241 [[nodiscard]] static bool validate_websocket_frame(
242 size_t payload_length,
243 size_t max_size = message_limits::MAX_WEBSOCKET_FRAME) noexcept {
244 return payload_length <= max_size;
245 }
246
253 [[nodiscard]] static validation_result validate_url(
254 std::string_view url) noexcept {
255 if (url.size() > message_limits::MAX_URL_LENGTH) {
256 return validation_result::size_exceeded;
257 }
258
259 if (url.find('\0') != std::string_view::npos) {
260 return validation_result::null_byte_detected;
261 }
262
263 return validation_result::ok;
264 }
265
275 [[nodiscard]] static bool contains_suspicious_pattern(
276 std::string_view data) noexcept {
277 // Check for NULL byte injection
278 if (data.find('\0') != std::string_view::npos) {
279 return true;
280 }
281
282 // Check for HTTP response splitting
283 if (data.find("\r\n\r\n") != std::string_view::npos) {
284 return true;
285 }
286
287 return false;
288 }

296 [[nodiscard]] static std::string sanitize_string(std::string_view input) {
297 std::string result;
298 result.reserve(input.size());
299
300 for (char c : input) {
301 // Keep printable ASCII and common whitespace
302 if (c >= 0x20 || c == '\t' || c == '\n' || c == '\r') {
303 result += c;
304 }
305 }
306
307 return result;
308 }
309
319 [[nodiscard]] static size_t safe_buffer_size(
320 size_t requested_size,
321 size_t max_size = message_limits::MAX_MESSAGE_SIZE) noexcept {
322 return std::min(requested_size, max_size);
323 }
324};
