102auto reverse_handler(grpc::server_context& ctx, const std::vector<uint8_t>& request)
103 -> std::pair<grpc::grpc_status, std::vector<uint8_t>>
104{
105 std::cout << "[Reverse] Processing " << request.size() << " bytes" << std::endl;
106
107 if (request.empty()) {
108 return {
109 grpc::grpc_status::error_status(
110 grpc::status_code::invalid_argument,
111 "Request cannot be empty"),
112 {}
113 };
114 }
115
116 std::vector<uint8_t> reversed(request.rbegin(), request.rend());
117 return {grpc::grpc_status::ok_status(), reversed};
118}

123auto timestamp_handler(grpc::server_context& ctx, const std::vector<uint8_t>& request)
124 -> std::pair<grpc::grpc_status, std::vector<uint8_t>>
125{
126 auto now = std::chrono::system_clock::now();
127 auto ms = std::chrono::duration_cast<std::chrono::milliseconds>(
128 now.time_since_epoch()).count();
129
130 std::string timestamp = std::to_string(ms);
131 std::vector<uint8_t> response(timestamp.begin(), timestamp.end());
132
133 std::cout << "[Timestamp] Returning: " << timestamp << std::endl;
134 return {grpc::grpc_status::ok_status(), response};
135}

140auto stream_handler(grpc::server_context& ctx,
141 const std::vector<uint8_t>& request,
142 grpc::server_writer& writer) -> grpc::grpc_status
143{
144 std::cout << "[Stream] Starting stream for " << request.size() << " byte request" << std::endl;
145
146 for (int i = 0; i < 5; ++i) {
147 std::string chunk = "Chunk " + std::to_string(i + 1);
148 std::vector<uint8_t> data(chunk.begin(), chunk.end());
149
150 auto result = writer.write(data);
151 if (result.is_err()) {
152 return grpc::grpc_status::error_status(
153 grpc::status_code::internal,
154 "Failed to write chunk " + std::to_string(i + 1));
155 }
156
157 std::cout << "[Stream] Sent: " << chunk << std::endl;
158 }
159
160 return grpc::grpc_status::ok_status();
161}

167void demo_service_registration()
168{
169 std::cout << "\n=== Service Registration Demo ===" << std::endl;
170
171 // Create registry with health check and reflection
172 grpc::registry_config config;
173 config.enable_health_check = true;
174 config.enable_reflection = true;
175
176 grpc::service_registry registry(config);
177
178 // Create Echo service
179 grpc::generic_service echo_service("demo.EchoService");
180 echo_service.register_unary_method("Echo", echo_handler, "EchoRequest", "EchoResponse");
181 echo_service.register_unary_method("Reverse", reverse_handler);
182
183 // Create Utility service
184 grpc::generic_service util_service("demo.UtilityService");
185 util_service.register_unary_method("GetTimestamp", timestamp_handler);
186 util_service.register_server_streaming_method("StreamData", stream_handler);
187
188 // Register services
189 auto result1 = registry.register_service(&echo_service);
190 auto result2 = registry.register_service(&util_service);
191
192 if (result1.is_ok() && result2.is_ok()) {
193 std::cout << "Successfully registered services:" << std::endl;
194 for (const auto& name : registry.service_names()) {
195 std::cout << " - " << name << std::endl;
196 }
197 }
198
199 // Verify method lookup
200 auto method = registry.find_method("/demo.EchoService/Echo");
201 if (method.has_value()) {
202 std::cout << "Found method: " << method->second->name
203 << " (type: " << static_cast<int>(method->second->type) << ")"
204 << std::endl;
205 }
206}

208void demo_message_framing()
209{
210 std::cout << "\n=== Message Framing Demo ===" << std::endl;
211
212 // Create a sample payload
213 std::string payload = "Hello, gRPC!";
214 std::vector<uint8_t> data(payload.begin(), payload.end());
215
216 // Create gRPC message
217 grpc::grpc_message msg(data, false);
218
219 std::cout << "Original data size: " << data.size() << " bytes" << std::endl;
220 std::cout << "Serialized size: " << msg.serialized_size() << " bytes" << std::endl;
221 std::cout << "Compressed: " << (msg.compressed ? "yes" : "no") << std::endl;
222
223 // Serialize
224 auto serialized = msg.serialize();
225
226 // Parse back
227 auto parsed = grpc::grpc_message::parse(serialized);
228 if (parsed.is_ok()) {
229 std::string recovered(parsed.value().data.begin(), parsed.value().data.end());
230 std::cout << "Recovered data: " << recovered << std::endl;
231 }
232}

234void demo_health_checking()

237
238 grpc::health_service health;
239
240 // Set initial status
241 health.set_status("demo.EchoService", grpc::health_status::serving);
242 health.set_status("demo.UtilityService", grpc::health_status::serving);
243 health.set_status("", grpc::health_status::serving); // Server-wide status
244
245 std::cout << "Initial health status:" << std::endl;
246 std::cout << " demo.EchoService: " << (health.get_status("demo.EchoService") == grpc::health_status::serving ? "SERVING" : "NOT_SERVING") << std::endl;
247 std::cout << " demo.UtilityService: " << (health.get_status("demo.UtilityService") == grpc::health_status::serving ? "SERVING" : "NOT_SERVING") << std::endl;
248
249 // Simulate service going down
250 health.set_status("demo.UtilityService", grpc::health_status::not_serving);
251
252 std::cout << "\nAfter UtilityService goes down:" << std::endl;
253 std::cout << " demo.EchoService: " << (health.get_status("demo.EchoService") == grpc::health_status::serving ? "SERVING" : "NOT_SERVING") << std::endl;
254 std::cout << " demo.UtilityService: " << (health.get_status("demo.UtilityService") == grpc::health_status::serving ? "SERVING" : "NOT_SERVING") << std::endl;
255
256 // Unknown service
257 auto unknown = health.get_status("unknown.Service");
258 std::cout << "\nUnknown service status: " << (unknown == grpc::health_status::service_unknown ? "SERVICE_UNKNOWN" : "OTHER") << std::endl;
259}

319void demo_handler_invocation()
320{
321 std::cout << "\n=== Handler Invocation Demo ===" << std::endl;
322
323 // Create service
324 grpc::generic_service service("demo.TestService");
325 service.register_unary_method("Echo", echo_handler);
326 service.register_unary_method("Reverse", reverse_handler);
327
328 // Simulate request
329 std::string request_data = "Test Data";
330 std::vector<uint8_t> request(request_data.begin(), request_data.end());
331
332 // Get and invoke handler
333 auto* handler = service.get_unary_handler("Echo");
334 if (handler) {
335 mock_server_context ctx;
336 auto [status, response] = (*handler)(ctx, request);
337
338 if (status.is_ok()) {
339 std::string result(response.begin(), response.end());
340 std::cout << "Echo result: " << result << std::endl;
341 }
342 }
343
344 handler = service.get_unary_handler("Reverse");
345 if (handler) {
346 mock_server_context ctx;
347 auto [status, response] = (*handler)(ctx, request);
348
349 if (status.is_ok()) {
350 std::string result(response.begin(), response.end());
351 std::cout << "Reverse result: " << result << std::endl;
352 }
353 }
354
355 // Non-existent handler
356 auto* missing = service.get_unary_handler("NonExistent");
357 std::cout << "NonExistent handler: " << (missing ? "found" : "not found") << std::endl;
358}
