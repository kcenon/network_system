42// RAII wrapper for automatic connection release
43class scoped_connection
44{
45public:
46 explicit scoped_connection(std::shared_ptr<connection_pool> pool)
47 : pool_(std::move(pool)), client_(pool_->acquire())
48 {
49 }
50
51 ~scoped_connection()
52 {
53 if (client_)
54 {
55 pool_->release(std::move(client_));
56 }
57 }
58
59 auto operator->() { return client_.get(); }
60 auto get() { return client_.get(); }
61
62 // Disable copy
63 scoped_connection(const scoped_connection&) = delete;
64 scoped_connection& operator=(const scoped_connection&) = delete;
65
66 // Enable move
67 scoped_connection(scoped_connection&& other) noexcept
68 : pool_(std::move(other.pool_)), client_(std::move(other.client_))
69 {
70 }
71
72 scoped_connection& operator=(scoped_connection&& other) noexcept
73 {
74 if (this != &other)
75 {
76 if (client_)
77 {
78 pool_->release(std::move(client_));
79 }
80 pool_ = std::move(other.pool_);
81 client_ = std::move(other.client_);
82 }
83 return *this;
84 }
85
86private:
87 std::shared_ptr<connection_pool> pool_;
88 std::unique_ptr<messaging_client> client_;
89};

155void concurrent_usage_example(std::shared_ptr<connection_pool> pool)
156{
157 std::cout << "=== Concurrent Usage Example ===\n";
158
159 constexpr int num_threads = 10;
160 constexpr int requests_per_thread = 5;
161
162 std::atomic<int> success_count{0};
163 std::atomic<int> failure_count{0};
164 std::vector<std::thread> workers;
165
166 auto start_time = std::chrono::steady_clock::now();
167
168 // Spawn worker threads
169 for (int i = 0; i < num_threads; ++i)
170 {
171 workers.emplace_back(
172 [pool, i, &success_count, &failure_count, requests_per_thread]()
173 {
174 for (int j = 0; j < requests_per_thread; ++j)
175 {
176 auto client = pool->acquire();
177
178 // Simulate some work
179 std::vector<uint8_t> data = {static_cast<uint8_t>('A' + i),
180 static_cast<uint8_t>('0' + j)};
181
182 auto result = client->send_packet(std::move(data));
183
184 if (result.is_ok())
185 {
186 success_count.fetch_add(1, std::memory_order_relaxed);
187 }
188 else
189 {
190 failure_count.fetch_add(1, std::memory_order_relaxed);
191 }
192
193 pool->release(std::move(client));
194
195 // Small delay to simulate processing
196 std::this_thread::sleep_for(std::chrono::milliseconds(10));
197 }
198 });
199 }
200
201 // Wait for all workers
202 for (auto& w : workers)
203 {
204 w.join();
205 }
206
207 auto end_time = std::chrono::steady_clock::now();
208 auto duration = std::chrono::duration_cast<std::chrono::milliseconds>(
209 end_time - start_time);
210
211 int total_requests = num_threads * requests_per_thread;
212 std::cout << "Completed " << success_count.load() << "/" << total_requests
213 << " requests successfully\n";
214 std::cout << "Failed: " << failure_count.load() << "\n";
215 std::cout << "Duration: " << duration.count() << "ms\n";
216 std::cout << "Throughput: "
217 << (total_requests * 1000.0 / duration.count()) << " req/s\n\n";
218}

306 // Store sessions for echo functionality

308 std::map<std::string, std::shared_ptr<interfaces::i_session>> sessions;
309
310 // Set up connection handler to track sessions
311 server->set_connection_callback(
312 [&sessions, &sessions_mutex](std::shared_ptr<interfaces::i_session> session)
313 {
314 std::lock_guard<std::mutex> lock(sessions_mutex);
315 sessions[std::string(session->id())] = session;
316 });
317
318 // Set up disconnection handler to clean up sessions
319 server->set_disconnection_callback(
320 [&sessions, &sessions_mutex](std::string_view session_id)
321 {
322 std::lock_guard<std::mutex> lock(sessions_mutex);
323 sessions.erase(std::string(session_id));
324 });
325
326 // Set up echo handler
327 server->set_receive_callback(
328 [&sessions, &sessions_mutex](std::string_view session_id, const std::vector<uint8_t>& data)
329 {
330 std::lock_guard<std::mutex> lock(sessions_mutex);
331 auto it = sessions.find(std::string(session_id));
332 if (it != sessions.end())
333 {
334 // Echo back the data
335 it->second->send(std::vector<uint8_t>(data));
336 }
337 });
338
339 auto server_result = server->start(5555);
340 if (server_result.is_err())
341 {
342 std::cerr << "Failed to start server: " << server_result.error().message
343 << "\n";
