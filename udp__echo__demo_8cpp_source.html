37void run_server()
38{
39 std::cout << "[Server] Starting UDP echo server...\n";
40
41 facade::udp_facade udp;
42 auto server = udp.create_server({.port = 5555, .server_id = "EchoServer"});
43
44 // Track sessions to send responses back
45 std::unordered_map<std::string, std::shared_ptr<interfaces::i_session>> sessions;
46 std::mutex sessions_mutex;
47
48 // Set up connection callback to track new clients
49 server->set_connection_callback(
50 [&sessions, &sessions_mutex](std::shared_ptr<interfaces::i_session> session)
51 {
52 std::lock_guard<std::mutex> lock(sessions_mutex);
53 std::string id(session->id());
54 sessions[id] = session;
55 std::cout << "[Server] New client: " << id << "\n";
56 });
57
58 // Set up disconnection callback
59 server->set_disconnection_callback(
60 [&sessions, &sessions_mutex](std::string_view session_id)
61 {
62 std::lock_guard<std::mutex> lock(sessions_mutex);
63 sessions.erase(std::string(session_id));
64 std::cout << "[Server] Client disconnected: " << session_id << "\n";
65 });
66
67 // Set up receive callback to echo messages back (using i_protocol_server interface)
68 server->set_receive_callback(
69 [&sessions, &sessions_mutex](std::string_view session_id,
70 const std::vector<uint8_t>& data)
71 {
72 std::string message(data.begin(), data.end());
73 std::cout << "[Server] Received: \"" << message << "\" from " << session_id << "\n";
74
75 // Echo back via session
76 std::shared_ptr<interfaces::i_session> session;
77 {
78 std::lock_guard<std::mutex> lock(sessions_mutex);
79 auto it = sessions.find(std::string(session_id));
80 if (it != sessions.end())
81 {
82 session = it->second;
83 }
84 }
85
86 if (session)
87 {
88 std::string response = "Echo: " + message;
89 std::vector<uint8_t> response_data(response.begin(), response.end());
90 auto result = session->send(std::move(response_data));
91 if (result.is_ok())
92 {
93 std::cout << "[Server] Sent echo response: " << response.size() << " bytes\n";
94 }
95 else
96 {
97 std::cerr << "[Server] Send error: " << result.error().message << "\n";
98 }
99 }
100 });
101
102 // Set up error callback
103 server->set_error_callback(
104 [](std::string_view session_id, std::error_code ec)
105 {
106 std::cerr << "[Server] Error on session " << session_id << ": " << ec.message() << "\n";
107 });
108
109 std::cout << "[Server] Running on port 5555. Press Ctrl+C to stop.\n";
110
111 // Wait until signaled to stop
112 while (!should_stop.load())
113 {
114 std::this_thread::sleep_for(std::chrono::milliseconds(100));
115 }
116
117 std::cout << "[Server] Stopping...\n";
118 std::cout << "[Server] Stopped.\n";
119}

126void run_client()
127{
128 // Give server time to start
129 std::this_thread::sleep_for(std::chrono::milliseconds(500));
130
131 std::cout << "[Client] Starting UDP client...\n";
132
133 facade::udp_facade udp;
134 auto client = udp.create_client({.host = "localhost", .port = 5555, .client_id = "TestClient"});
135
136 // Set up receive callback to handle echo responses (using i_protocol_client interface)
137 client->set_receive_callback(
138 [](const std::vector<uint8_t>& data)
139 {
140 std::string message(data.begin(), data.end());
141 std::cout << "[Client] Received response: \"" << message << "\"\n";
142 });
143
144 // Set up error callback
145 client->set_error_callback(
146 [](std::error_code ec)
147 {
148 std::cerr << "[Client] Error: " << ec.message() << "\n";
149 });
150
151 std::cout << "[Client] Connected to localhost:5555\n";
152
153 // Send a few test messages
154 const std::vector<std::string> messages = {
155 "Hello, UDP!",
156 "This is a test message",
157 "UDP is fast!",
158 "Final message"
159 };
160
161 for (const auto& msg : messages)
162 {
163 std::vector<uint8_t> data(msg.begin(), msg.end());
164
165 std::cout << "[Client] Sending: \"" << msg << "\"\n";
166
167 // Use i_protocol_client::send() - no callback support in unified interface
168 auto send_result = client->send(std::move(data));
169
170 if (send_result.is_ok())
171 {
172 std::cout << "[Client] Sent " << msg.size() << " bytes\n";
173 }
174 else if (send_result.is_err())
175 {
176 std::cerr << "[Client] Send failed: " << send_result.error().message << "\n";
177 }
178
179 // Wait between messages
180 std::this_thread::sleep_for(std::chrono::milliseconds(1000));
181 }
182
183 // Wait for responses
184 std::this_thread::sleep_for(std::chrono::milliseconds(1000));
185
186 std::cout << "[Client] Stopping...\n";
187 std::cout << "[Client] Stopped.\n";
188
189 // Signal server to stop
190 should_stop.store(true);
191}
