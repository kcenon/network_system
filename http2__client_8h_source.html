78 struct http2_stream

84 std::vector<uint8_t> request_body;
85 std::vector<uint8_t> response_body;
86 int32_t window_size = 65535;
87 std::promise<http2_response> promise;
88 bool headers_complete = false;
89 bool body_complete = false;
90
92 bool is_streaming = false;
93 std::function<void(std::vector<uint8_t>)> on_data;
94 std::function<void(std::vector<http_header>)> on_headers;
95 std::function<void(int)> on_complete;
96
97 http2_stream() = default;
98 http2_stream(http2_stream&& other) noexcept { *this = std::move(other); }
99 auto operator=(http2_stream&& other) noexcept -> http2_stream&
100 {
101 if (this == &other) return *this;
102 stream_id = other.stream_id;
103 state.store(other.state.load());
104 request_headers = std::move(other.request_headers);
105 response_headers = std::move(other.response_headers);
106 request_body = std::move(other.request_body);
107 response_body = std::move(other.response_body);
108 window_size = other.window_size;
109 promise = std::move(other.promise);
110 headers_complete = other.headers_complete;
111 body_complete = other.body_complete;
112 is_streaming = other.is_streaming;
113 on_data = std::move(other.on_data);
114 on_headers = std::move(other.on_headers);
115 on_complete = std::move(other.on_complete);
116 return *this;
117 }
